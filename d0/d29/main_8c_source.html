456ColRGBA * get_color_from_hexa_string (gchar * color_string)
457{
458 ColRGBA * col = NULL;
459 const char * color = color_string;
460 // Ignorer le '#' si présent au début
461 if (color[0] == '#') color ++;
462
463 size_t len = strlen(color);
464 if (len != 3 && len != 4 && len != 6 && len != 8)
465 {
466 // longueur de chaîne incompatible
467 return NULL;
468 }
469 // Format long ?
470 int is_long = (len == 6 || len == 8) ? TRUE : FALSE;
471 float ratio = (is_long) ? 255.0 : 15.0;
472 int incr = (is_long) ? 2 : 1;
473 col = g_malloc0(sizeof*col);
474 col -> alpha = 1.0;
475 char buf[3] = {0}; // pour les caractères à examiner
476 // Format court (#RGB ou #RGBA)
477 buf[incr] = '\0';
478 buf[0] = color[0];
479 if (is_long) buf[1] = color[1];
480 col->red = strtol(buf, NULL, 16) / ratio;
481 buf[0] = color[incr];
482 if (is_long) buf[1] = color[incr+1];
483 col->green = strtol(buf, NULL, 16) / ratio;
484 buf[0] = color[2*incr];
485 if (is_long) buf[1] = color[2*incr+1];
486 col->blue = strtol(buf, NULL, 16) / ratio;
487 if (len == 4 || len == 8)
488 {
489 buf[0] = color[3*incr];
490 if (is_long) buf[1] = color[3*incr+1];
491 col->alpha = strtol(buf, NULL, 16) / ratio;
492 }
493
494 // Safety net
495 if (col -> red < 0.0 || col -> red > 1.0 ||
496 col -> green < 0.0 || col -> green > 1.0 ||
497 col -> blue < 0.0 || col -> blue > 1.0 ||
498 col -> alpha < 0.0 || col -> alpha > 1.0)
499 {
500 g_free(col);
501 return NULL;
502 }
503 return col;
504}

534int check_for_atomes_file_options (int start, int end, char *argv[])
535{
536 int i, j, k;
537 i=0;
538 for (j=start; j<end; j++)
539 {
540 k = test_this_arg (argv[j]);
541 if (k > 0 && j < end-1)
542 {
543 if (! (k == 1 && with_workspace))
544 {
545 if (! flist)
546 {
547 flist = g_malloc0(sizeof*flist);
548 ftmp = flist;
549 }
550 else
551 {
552 ftmp -> next = g_malloc0(sizeof*ftmp -> next);
553 ftmp = ftmp -> next;
554 }
555 ftmp -> file_name = g_strdup_printf ("%s", argv[j+1]);
556 ftmp -> file_type = k;
557 argv[j] = argv[j+1] = g_strdup_printf (" ");
558 j ++;
559 i ++;
560 if (k == 1) with_workspace = TRUE;
561 }
562 }
563 }
564 return i;
565}

589int parse_command_line (int argc, char *argv[])
590{
591 struct option atomes_options[] = {{"help", no_argument, 0, 'h'},
592 {"version", no_argument, 0, 'v'},
593 {"libreoffice", no_argument, 0, 'l'},
594 {"render-png", no_argument, 0, 'p'},
595 {"png", no_argument, 0, 'p'},
596 {"render-jpg", no_argument, 0, 'j'},
597 {"jpg", no_argument, 0, 'j'},
598 {"width", required_argument, 0, 'W'},
599 {"height", required_argument, 0, 'H'},
600 {"output", required_argument, 0, 'o'},
601 {"style", required_argument, 0, 's'},
602 {"atoms", required_argument, 0, 'e'},
603 {"poly", required_argument, 0, 't'},
604 {"axis", required_argument, 0, 'a'},
605 {"box", required_argument, 0, 'b'},
606 {"box_color", required_argument, 0, 'C'},
607 {"back_color", required_argument, 0, 'B'},
608 {"back_grad", required_argument, 0, 'G'},
609 {"back_dir", required_argument, 0, 'D'},
610 {"back_pos", required_argument, 0, 'P'},
611 {"grad_col_a", required_argument, 0, 'U'},
612 {"grad_col_b", required_argument, 0, 'V'},
613 {"rep", required_argument, 0, 'r'},
614 {"convert", required_argument, 0, 'c'},
615 // {"debug", no_argument, 0, 'd'},
616 {0, 0, 0, 0}};
617 int opt;
618 int index = -1;
619 int i, j;
620 double v;
621 gchar * image_x = NULL;
622 gchar * image_y = NULL;
623 int img_opt = 0;
624
625 for (i=0; i<2; i++) render_image_grad_color[i] = NULL;
626 /* We want to parse the command line using GNU getopt, but then
627 some options that are atomes specific and not GNU getopt compatible
628 might disappear, to prevent that it is mandatory to prepare
629 the information to be given to GNU getopt */
630 int files_to_read = check_for_atomes_file_options (1, argc, argv);
631
632 /* Letter follow by : means that the command requires an argument
633 No letter if the option is only in long format, ex : --width
634 If the long name is empty the command is only in short format */
635 while ((opt = getopt_long(argc, argv, "hvlpjdW:H:o:s:a:b:c:r:e:t:B:C:G:D:P:U:V:", atomes_options, & index)) != -1)
636 {
637 switch (opt)
638 {
639 case 'h':
640 set_atomes_locale ();
641 print_help();
642 return FALSE;
643 break;
644 case 'v':
645 set_atomes_locale ();
646 print_version();
647 return FALSE;
648 break;
649 case 'e':
650 render_image_acolor = get_color_map_from_string (g_ascii_strdown(optarg,strlen(optarg)));
651 trigger_fragment_analysis = (render_image_acolor > 2) ? TRUE : FALSE;
652 trigger_molecule_analysis = (render_image_acolor > 3) ? TRUE : FALSE;
653 img_opt ++;
654 img_opt += (index == -1) ? 1 : 0;
655 break;
656 case 't':
657 render_image_pcolor = get_color_map_from_string (g_ascii_strdown(optarg,strlen(optarg)));
658 trigger_fragment_analysis = (render_image_pcolor > 2) ? TRUE : FALSE;
659 trigger_molecule_analysis = (render_image_pcolor > 3) ? TRUE : FALSE;
660 img_opt ++;
661 img_opt += (index == -1) ? 1 : 0;
662 break;
663 case 'l':
664 atomes_from_libreoffice = TRUE;
665 render_image_format = 0;
666 break;
667 case 'p':
668 atomes_render_image = TRUE;
669 render_image_format = 0;
670 img_opt ++;
671 break;
672 case 'j':
673 atomes_render_image = TRUE;
674 render_image_format = 1;
675 img_opt ++;
676 break;
677 case 'o':
678 render_image_output = g_strdup_printf ("%s", optarg);
679 img_opt ++;
680 img_opt += (index == -1) ? 1 : 0;
681 break;
682 case 'W':
683 image_x = g_strdup_printf ("%s", optarg);
684 img_opt ++;
685 img_opt += (index == -1) ? 1 : 0;
686 break;
687 case 'H':
688 image_y = g_strdup_printf ("%s", optarg);
689 img_opt ++;
690 img_opt += (index == -1) ? 1 : 0;
691 break;
692 case 's':
693 render_image_style = get_style_from_string (g_ascii_strdown(optarg,strlen(optarg)));
694 img_opt ++;
