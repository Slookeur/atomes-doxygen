458ColRGBA * get_color_from_hexa_string (gchar * color_string)
459{
460 ColRGBA * col = NULL;
461 const char * color = color_string;
462 // Ignorer le '#' si présent au début
463 if (color[0] == '#') color ++;
464
465 size_t len = strlen(color);
466 if (len != 3 && len != 4 && len != 6 && len != 8)
467 {
468 // longueur de chaîne incompatible
469 return NULL;
470 }
471 // Format long ?
472 int is_long = (len == 6 || len == 8) ? TRUE : FALSE;
473 float ratio = (is_long) ? 255.0 : 15.0;
474 int incr = (is_long) ? 2 : 1;
475 col = g_malloc0(sizeof*col);
476 col -> alpha = 1.0;
477 char buf[3] = {0}; // pour les caractères à examiner
478 // Format court (#RGB ou #RGBA)
479 buf[incr] = '\0';
480 buf[0] = color[0];
481 if (is_long) buf[1] = color[1];
482 col->red = strtol(buf, NULL, 16) / ratio;
483 buf[0] = color[incr];
484 if (is_long) buf[1] = color[incr+1];
485 col->green = strtol(buf, NULL, 16) / ratio;
486 buf[0] = color[2*incr];
487 if (is_long) buf[1] = color[2*incr+1];
488 col->blue = strtol(buf, NULL, 16) / ratio;
489 if (len == 4 || len == 8)
490 {
491 buf[0] = color[3*incr];
492 if (is_long) buf[1] = color[3*incr+1];
493 col->alpha = strtol(buf, NULL, 16) / ratio;
494 }
495
496 // Safety net
497 if (col -> red < 0.0 || col -> red > 1.0 ||
498 col -> green < 0.0 || col -> green > 1.0 ||
499 col -> blue < 0.0 || col -> blue > 1.0 ||
500 col -> alpha < 0.0 || col -> alpha > 1.0)
501 {
502 g_free(col);
503 return NULL;
504 }
505 return col;
506}

536int check_for_atomes_file_options (int start, int end, char *argv[])
537{
538 int i, j, k;
539 i=0;
540 for (j=start; j<end; j++)
541 {
542 k = test_this_arg (argv[j]);
543 if (k > 0 && j < end-1)
544 {
545 if (! (k == 1 && with_workspace))
546 {
547 if (! flist)
548 {
549 flist = g_malloc0(sizeof*flist);
550 ftmp = flist;
551 }
552 else
553 {
554 ftmp -> next = g_malloc0(sizeof*ftmp -> next);
555 ftmp = ftmp -> next;
556 }
557 ftmp -> file_name = g_strdup_printf ("%s", argv[j+1]);
558 ftmp -> file_type = k;
559 argv[j] = argv[j+1] = g_strdup_printf (" ");
560 j ++;
561 i ++;
562 if (k == 1) with_workspace = TRUE;
563 }
564 }
565 }
566 return i;
567}

677 img_opt ++;
678 break;
679 case 'o':
