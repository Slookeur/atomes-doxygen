200void update_error_trace (const char * file, const char * func, int trace_line)
201{

203 gchar * file_part;
204 gchar * func_part;

282 }

284 {
285 gchar * err;
286 if (pfile != NULL && ! wid)
287 {

290 "\tError %s %s\n"
291 "\tProject file version: <b>%1.1f</b>\n"),
292 (! act) ? _("open") : _("save"),
293 pfile,
294 (! act) ? _("reading") : _("saving"),
295 _(project_error -> error_signal.message),
296 project_file_version);
297
298 }
299 else if (wid)
300 {
301 err = g_strdup_printf (_("Impossible to %s workspace file.\n"
302 "Error with project: \n\n"
303 "\t\t%s\n\n"
304 "\tError %s %s\n"
